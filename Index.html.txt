            addMarkRow("Unit Test 1", "Cell Biology & Cytology", 50, 0);

        function updateCalculations() {
            // Get Student Meta Values
            const name = document.getElementById('studentName').value.trim() || '[Student Name]';
            const roll = document.getElementById('rollNo').value.trim() || '[Roll No]';
            const sClass = document.getElementById('studentClass').value;
            const session = document.getElementById('academicSession').value.trim() || '2025 - 2026';
            const guardian = document.getElementById('guardianName').value.trim() || 'N/A';
            const rawDate = document.getElementById('reportDate').value;

            let formattedDate = 'N/A';
            if (rawDate) {
                const d = new Date(rawDate);
                formattedDate = d.toLocaleDateString('en-GB', { day: 'numeric', month: 'short', year: 'numeric' });
            }

            // Sync Header Details
            document.getElementById('docName').textContent = name;
            document.getElementById('docRoll').textContent = roll;
            document.getElementById('docClass').textContent = sClass;
            document.getElementById('docSession').textContent = session;
            document.getElementById('docGuardian').textContent = guardian;
            document.getElementById('docDate').textContent = formattedDate;

            // Compute Sum Totals
            const markItems = document.querySelectorAll('#marksContainer > div');
            let totalMax = 0;
            let totalObt = 0;

            const docTableBody = document.getElementById('docTableBody');
            docTableBody.innerHTML = '';

            markItems.forEach(item => {
                const exam = item.querySelector('.exam-input').value.trim() || 'General Test';
                const topic = item.querySelector('.topic-input').value.trim() || 'Biology Chapter';
                const max = parseFloat(item.querySelector('.max-input').value) || 0;
                let obt = parseFloat(item.querySelector('.obt-input').value) || 0;

                if (obt > max) {
                    obt = max;
                    item.querySelector('.obt-input').value = max;
                }

                totalMax += max;
                totalObt += obt;

                const percent = max > 0 ? ((obt / max) * 100).toFixed(1) : '0.0';

                // Render Live Table Row
                const tr = document.createElement('tr');
                tr.className = "hover:bg-slate-50 transition";
                tr.innerHTML = `
                    <td class="p-2 border border-slate-200 font-semibold text-slate-800">${exam}</td>
                    <td class="p-2 border border-slate-200 text-slate-600">${topic}</td>
                    <td class="p-2 border border-slate-200 text-center">${max}</td>
                    <td class="p-2 border border-slate-200 text-center font-bold text-emerald-900">${obt}</td>
                    <td class="p-2 border border-slate-200 text-center font-medium">${percent}%</td>
                `;
                docTableBody.appendChild(tr);
            });

            // Calculate Percentage & Auto Grade
            const percentage = totalMax > 0 ? (totalObt / totalMax) * 100 : 0;
            const formattedPercentage = percentage.toFixed(1);

            let grade = 'F';
            let remark = 'Requires immediate attention and dedicated practice sessions. Regular doubt-clearing recommended.';
            let status = 'Needs Improvement';

            if (percentage >= 90) {
                grade = 'A+ (Outstanding)';
                remark = 'Outstanding conceptual mastery in Biology! Exceptional performance across Botany & Zoology modules.';
                status = 'Passed with Distinction';
            } else if (percentage >= 80) {
                grade = 'A';
                remark = 'Excellent understanding of core topics. Keep practicing NEET level numericals and diagrammatic questions.';
                status = 'First Class Distinction';
            } else if (percentage >= 70) {
                grade = 'B+';
                remark = 'Good effort! Focus on thorough line-by-line reading of NCERT textbooks and regular practice papers.';
                status = 'Passed (First Division)';
            } else if (percentage >= 60) {
                grade = 'B';
                remark = 'Satisfactory performance. Consistent daily revision required to strengthen high-weightage chapters.';
                status = 'Passed (Second Division)';
            } else if (percentage >= 50) {
                grade = 'C';
                remark = 'Average performance. Needs targeted doubt sessions and regular attempt of Daily Practice Problems (DPPs).';
                status = 'Passed';
            }

            if (!isRemarkUserModified) {
                document.getElementById('customRemark').value = remark;
            }

            const activeRemark = document.getElementById('customRemark').value;

            // Sync KPIs
            document.getElementById('kpiTotal').textContent = `${totalObt} / ${totalMax}`;
            document.getElementById('kpiPercent').textContent = `${formattedPercentage}%`;
            document.getElementById('kpiGrade').textContent = grade.split(' ')[0];
            document.getElementById('kpiStatus').textContent = status;

            // Sync Document Card
            document.getElementById('docScore').textContent = `${totalObt} / ${totalMax}`;
            document.getElementById('docPercent').textContent = `${formattedPercentage}%`;
            document.getElementById('docGrade').textContent = grade;
            document.getElementById('docRemark').textContent = activeRemark;
        }
